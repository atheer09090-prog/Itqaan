function pickWeightedOutcome() {

// 55% valid, 45% invalid (split among 3 reasons by weight)

const validChance = 55;

const roll = Math.random() * 100;

if (roll < validChance) return null; // valid


// Pick invalid reason

const totalWeight = INVALID_REASONS.reduce((s, r) => s + r.weight, 0);

let r = Math.random() * totalWeight;

for (const reason of INVALID_REASONS) {

r -= reason.weight;

if (r <= 0) return reason;

}

return INVALID_REASONS[0];

}

function buildDonutChart(approved, rejected, pending, fraud, total) {

if (total === 0) return '<div style="text-align:center;color:var(--text-muted);padding:20px;">لا توجد بيانات</div>';

const cx = 60, cy = 60, r = 44, sw = 13;

const circ = 2 * Math.PI * r;

const segments = [

{ label: 'مقبول', val: approved, color: '#0D7A50' },

{ label: 'مرفوض', val: rejected, color: '#B91C1C' },

{ label: 'قيد المراجعة', val: pending, color: '#B45309' },

{ label: 'مشبوه', val: fraud, color: '#7C3AED' },

].filter(s => s.val > 0);


let offset = 0;

const paths = segments.map(s => {

const dash = (s.val / total) * circ;

const path = `<circle cx="${cx}" cy="${cy}" r="${r}" fill="none" stroke="${s.color}"

stroke-width="${sw}" stroke-dasharray="${dash.toFixed(2)} ${(circ - dash).toFixed(2)}"

stroke-dashoffset="${(-offset).toFixed(2)}"/>`;

offset += dash;

return path;

}).join('');


const legend = segments.map(s => `

<div class="legend-item">

<div class="legend-dot" style="background:${s.color};"></div>

<span class="legend-label">${s.label}</span>

<span class="legend-val">${s.val}</span>

</div>`).join('');


return `

<div class="donut-wrap">

<svg width="120" height="120" viewBox="0 0 120 120" style="flex-shrink:0;transform:rotate(-90deg);">

<circle cx="${cx}" cy="${cy}" r="${r}" fill="none" stroke="var(--bg-panel)" stroke-width="${sw}"/>

<text x="${cx}" y="${cy + 6}" text-anchor="middle" font-size="15" font-weight="900"

</div>`;

}

]);


const BOM = '\uFEFF';

const csv = BOM + [headers, ...rows].map(r => r.join(',')).join('\n');

const blob = new Blob([csv], { type: 'text/csv;charset=utf-8;' });

const url = URL.createObjectURL(blob);

a.download = `تقرير-التحقق-العقاري-${new Date().toISOString().slice(0, 10)}.csv`;

document.body.appendChild(a);

a.click();

document.body.removeChild(a);

URL.revokeObjectURL(url);

showToast('✅ تم تصدير التقرير بنجاح');

}
